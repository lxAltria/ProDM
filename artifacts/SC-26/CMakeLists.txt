# SC'26 (IPComp comparison): parallel drivers of the multilevel (PMGARD-HB), proaicd and PSZ3-delta pipelines.
if(NOT PRODM_WITH_MPI)
    return()
endif()
find_package(MPI)
if(NOT MPI_FOUND)
    message(STATUS "Skipping the SC-26 parallel tools (MPI not found)")
    return()
endif()
function(add_mpi_executable target source)
    add_executable(${target} ${source})
    target_link_libraries(${target} ${PROJECT_NAME} MPI::MPI_CXX)
endfunction()
add_mpi_executable(para_mdr_refactor parallel/para_mdr_refactor.cpp)
add_mpi_executable(para_mdr_reconstructor parallel/para_mdr_reconstructor.cpp)
add_mpi_executable(para_proaicd_refactor parallel/para_proaicd_refactor.cpp)
add_mpi_executable(para_proaicd_reconstructor parallel/para_proaicd_reconstructor.cpp)
if(PRODM_WITH_SZ3)
    add_mpi_executable(para_PSZ3-delta_refactor parallel/para_PSZ3-delta_refactor.cpp)
    add_mpi_executable(para_PSZ3-delta_reconstructor parallel/para_PSZ3-delta_reconstructor.cpp)
else()
    message(STATUS "Skipping para_PSZ3-delta tools (require PRODM_WITH_SZ3)")
endif()
